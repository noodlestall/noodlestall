ingredientGroups:
  - title: Ingredients
    ingredients:
      - name: Spaghetti
        quantity: 450
        unit: g
      - name: Olive oil
        quantity: 120
        unit: ml
      - name: Garlic cloves
        quantity: 6
        unit: piece
        note: thinly sliced
      - name: Chilli flakes
        note: ¼ teaspoon, or to taste
      - name: Salt and freshly ground black pepper
        note: to taste
      - name: Flat-leaf parsley
        note: ¼ cup, chopped
      - name: Parmigiano-Reggiano
        note: 1 cup, finely grated
instructions:
  - title: Cook the spaghetti
    text: Boil the spaghetti in lightly salted water until just firm to the bite, 10 to 12 minutes. Drain and tip into a large bowl.
  - title: Toast the garlic
    text: Meanwhile put the oil and garlic in a cold frying pan over medium heat. Lower the heat once the oil bubbles and cook, stirring, until the garlic is golden, about 15 minutes in all. Take off the heat.
  - title: Season the pasta
    text: Stir the chilli flakes, salt and pepper through the spaghetti.
  - title: Toss
    text: Pour over the hot garlic oil, add the parsley and half the cheese and toss until coated.
  - title: Serve
    text: Serve topped with the remaining cheese.
