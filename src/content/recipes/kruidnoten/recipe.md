ingredientGroups:
  - title: Ingredients
    ingredients:
      - name: Unsalted butter
        quantity: 150
        unit: g
      - name: Dark soft brown sugar
        quantity: 125
        unit: g
      - name: Speculaas spice mix
        note: 2 tablespoons
      - name: Self-raising flour
        quantity: 250
        unit: g
        note: or plain flour plus 1½ teaspoons baking powder
      - name: Salt
        quantity: 3
        unit: g
      - name: Milk
        quantity: 60
        unit: ml
instructions:
  - title: Heat the oven
    text: Heat the oven to 160°C and line two baking trays with baking paper.
  - title: Make the dough
    text: Knead the butter, sugar, spice mix, flour and salt together. Add the milk a spoonful at a time until you have a supple dough.
  - title: Roll and cut
    text: Cut the dough into 4 pieces and roll each into a long rope about 1 cm thick. Cut the ropes into 1 cm pieces and roll each into a ball.
  - title: Shape
    text: Lay the balls on the trays and press each one slightly flat.
  - title: Bake
    text: Bake for about 15 minutes until cooked through and lightly browned, swapping the trays halfway. Cool for 15 minutes before eating.
