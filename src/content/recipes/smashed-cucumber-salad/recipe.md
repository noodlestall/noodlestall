tags:

ingredientGroups:
  - title: Ingredients
    ingredients:
      - name: Seedless cucumbers
        quantity: 2
        unit: piece
        note: English type, about 500 to 700 g in total
      - name: Salt
        quantity: 5
        unit: g
      - name: Sugar
        quantity: 10
        unit: g
      - name: Sesame oil
        quantity: 10
        unit: ml
      - name: Light soy sauce
        quantity: 15
        unit: ml
      - name: Rice vinegar
        quantity: 22.5
        unit: ml
      - name: Garlic cloves
        quantity: 3
        unit: piece
        note: finely chopped, use 2 to 4 to taste
      - name: Chilli oil
        quantity: 5
        unit: ml
        note: up to double, to taste
        optional: true
      - name: Toasted sesame seeds
        quantity: 5
        unit: g
      - name: Coriander
        quantity: 5
        unit: g
        note: chopped
instructions:
  - title: Make the dressing
    text: Wash and dry the cucumbers. Whisk the salt, sugar, sesame oil, soy sauce and rice vinegar until the salt and sugar dissolve.
  - title: Smash the cucumbers
    text: Lay a wide knife flat on a cucumber and press down until it cracks into long pieces. Work along its length, then cut the pieces on the diagonal into bite-sized chunks.
  - title: Toss and serve
    text: Toss the cucumber with the dressing, garlic and chilli oil in a large bowl. Serve at once with the sesame seeds and coriander on top. Without chilli oil, drizzle over a spoonful of hot neutral oil instead.
